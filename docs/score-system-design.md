## ២. គោលការណ៍ស្នូល

> **តារាងពិន្ទុគឺជា data មិនមែន code។**

មុខងារពិន្ទុមិនគួរដឹងថា "បឋមសិក្សា" ជាអ្វីទេ។ វាគួរតែសួរថា៖ *ថ្នាក់នេះមានមុខវិជ្ជាអ្វីខ្លះ?* រួចបង្ហាញ grid តាមចម្លើយ។ ការឆ្លើយសំណួរនោះជាការងាររបស់ **Score Template Resolver**។

### ៣ ស្រទាប់ (3-Layer Resolution)

```
ស្រទាប់ ១ ── System Template      KruSmart seed តាមកម្រិត/ថ្នាក់ (read-only)
                  ↓ inherit
ស្រទាប់ ២ ── School Override      នាយកសាលាកែសម្រាប់សាលាខ្លួន
                  ↓ inherit
ស្រទាប់ ៣ ── Class Override       គ្រូកែសម្រាប់ថ្នាក់ខ្លួន  ← "customize ខ្លួនឯង"
                  ↓
            Effective Template     អ្វីដែល grid បង្ហាញពិតប្រាកដ
```

ស្រទាប់ខាងក្រោមមិន **copy** ស្រទាប់ខាងលើទេ — វា **inherit** រួច store តែ *delta* (បន្ថែម / លាក់ / ប្តូរឈ្មោះ / ប្តូរមេគុណ)។ ហេតុផល៖ ពេលក្រសួងប្តូរកម្មវិធីសិក្សា អ្នកកែ system template មួយកន្លែង ហើយសាលាទាំងអស់ដែលមិនបាន override នឹងទទួលបានភ្លាម។

### ការធ្លាក់ចុះសម្រាប់គណនីចាស់ (Legacy Fallback)

នេះជាចំណុចដែល**មិនអាចខកខានបាន**។ តាម `CLAUDE.md` គណនីដែលគ្មាន `teacher_assignments` ត្រូវតែដំណើរការដដែល។ ដូច្នេះ៖

```
មាន class_id ?  ──ទេ──→  System Template "បឋមសិក្សា"
      │                    (= បញ្ជី hardcoded សព្វថ្ងៃ ដូចបេះបិទ)
      ├──បាទ──→ class → grade → education_level → resolve ៣ ស្រទាប់
```

System template បឋមសិក្សាត្រូវ seed ឲ្យ**ដូច array ក្នុង `ScoreEnterClient.tsx` សព្វថ្ងៃ ១០០%** — key ដដែល, label ដដែល, លំដាប់ដដែល។ បើធ្វើដូច្នេះ គណនីចាស់នឹងមិនដឹងថាមានអ្វីប្តូរទាល់តែសោះ ហើយ `scores.subject` ចាស់នៅតែ resolve ដដែល។

## ៣. Data Model

### 3.1 តារាងថ្មី: `score_template_subjects`

ជំនួស hardcoded array។ មួយ row = មួយជួរឈរក្នុង grid។

```sql
CREATE TABLE public.score_template_subjects (
    id                 UUID DEFAULT gen_random_uuid() PRIMARY KEY,

    -- scope: តើ row នេះជាស្រទាប់ណា? មួយក្នុងចំណោម ៣ ត្រូវតែ NOT NULL
    scope              TEXT NOT NULL CHECK (scope IN ('system','school','class')),
    education_level_id UUID REFERENCES public.education_levels(id) ON DELETE CASCADE,
    grade_id           UUID REFERENCES public.grades(id) ON DELETE CASCADE,
    school_id          UUID REFERENCES public.schools(id) ON DELETE CASCADE,
    class_id           UUID REFERENCES public.classes(id) ON DELETE CASCADE,

    -- អត្តសញ្ញាណ: នេះជាអ្វីដែលចូល scores.subject
    subject_key        TEXT NOT NULL,          -- 'math_general', 'sec_physics'
    label_km           TEXT NOT NULL,
    group_label        TEXT,                   -- ក្បាលក្រុមក្នុង picker

    -- លក្ខណៈពិន្ទុ — ផ្ទុកតែ max_score។ មេគុណ = max_score ÷ coefficientUnit
    -- មិនផ្ទុក coefficient ជា column ព្រោះវានឹង drift ពី max_score
    max_score          NUMERIC NOT NULL DEFAULT 10,
    score_types        TEXT[] DEFAULT ARRAY['monthly','semester'],
    columns            JSONB,                  -- ជួរឈររង (ស្តាប់/និយាយ/អាន/សរសេរ)
    value_kind         TEXT DEFAULT 'numeric'  -- 'numeric' | 'text' (sem_eval_*)
                       CHECK (value_kind IN ('numeric','text')),

    -- delta សម្រាប់ស្រទាប់ ២ និង ៣
    hidden             BOOLEAN DEFAULT false,  -- លាក់មុខវិជ្ជាដែល inherit មក
    sort_order         INTEGER DEFAULT 0,
    source             TEXT DEFAULT 'system'   -- 'system' | 'custom'
                       CHECK (source IN ('system','custom')),

    created_by         UUID REFERENCES auth.users(id),
    created_at         TIMESTAMPTZ DEFAULT now() NOT NULL,

    -- មួយ subject_key មិនអាចលេចឡើងពីរដងក្នុង scope តែមួយ
    UNIQUE NULLS NOT DISTINCT (scope, education_level_id, grade_id, school_id, class_id, subject_key)
);
```

**មិនប៉ះ `scores`។** `scores.subject` នៅតែជា TEXT key ដដែល — ទិន្នន័យចាស់ទាំងអស់នៅដដែល ហើយ `onConflict: 'teacher_id, student_id, subject, score_type, score_period'` មិនប្តូរ។ នេះជាមូលហេតុដែល design នេះមានតម្លៃទាប៖ វាប្តូរតែ *របៀបសម្រេចថាបង្ហាញជួរឈរណា* មិនមែន *របៀបស្តុកពិន្ទុ*។

### 3.2 មេគុណជាលទ្ធផល មិនមែនជា input

ក្បួន: **៥០ ពិន្ទុ = មេគុណ ១**

```
មេគុណ = ពិន្ទុពេញ ÷ ៥០

រូបវិទ្យា ទី១២   ពិន្ទុពេញ 75  → មេគុណ 1.5
មុខវិជ្ជា /100                 → មេគុណ 2
មុខវិជ្ជា /50                  → មេគុណ 1
មុខវិជ្ជា /25                  → មេគុណ 0.5
```

ដូច្នេះ **កុំផ្ទុក `coefficient` ជា column**។ ផ្ទុកតែ `max_score` រួចគណនាមេគុណ។ បើផ្ទុកទាំងពីរ ថ្ងៃណាមួយនឹងមានជួរដែល `max_score=75, coefficient=2` ហើយគ្មានអ្នកដឹងថាមួយណាត្រូវ។

តួលេខ ៥០ ជាក់ខ្លួនវាទៅជា config កម្រិត មិនមែន hardcode:

```ts
export interface GradingSchemeConfig {
  maxScore: number
  passMark: number
  bands: GradeBand[]
  weighting: 'simple' | 'coefficient'   // ថ្មី
  coefficientUnit?: number              // ថ្មី — ៥០ សម្រាប់អនុ/វិទ្យាល័យ
}

export const coefficientOf = (maxScore: number, cfg: GradingSchemeConfig) =>
  cfg.weighting === 'coefficient' ? maxScore / (cfg.coefficientUnit ?? 50) : 1
```

| កម្រិត | weighting | coefficientUnit | មធ្យមភាគចេញជា |
|---|---|---|---|
| បឋមសិក្សា | `simple` | — | /១០ |
| អនុវិទ្យាល័យ | `coefficient` | 50 | /៥០ |
| វិទ្យាល័យ | `coefficient` | 50 | /៥០ |

មធ្យមភាគ = `Σពិន្ទុទទួលបាន ÷ Σមេគុណ`។ សិស្សពិន្ទុពេញគ្រប់មុខវិជ្ជាបាន `Σ(មេគុណ×50) ÷ Σមេគុណ = 50` — ដូច្នេះមធ្យមភាគអនុ/វិទ្យាល័យចេញជា **/៥០** ដោយស្វ័យប្រវត្តិ គ្មានការធ្វើ normalize ដោយដៃ។

### 3.3 ថ្នាក់ពិន្ទុ (bands) ប្រើ **ភាគរយ** — តែស្លាកខុសគ្នាតាមកម្រិត

> ⚠️ **កែតាមឯកសារយោង kp-tralach.org (សីហា ២០២៦)។** ខ្ញុំធ្លាប់សរសេរថា "ជញ្ជាំងតែមួយសម្រាប់គ្រប់កម្រិត"។ នោះត្រូវតែពាក់កណ្តាល៖ **កម្រិតភាគរយដូចគ្នា តែស្លាកនិទ្ទេសខុសគ្នា** ហើយ **បាក់ឌុបមានតារាងផ្ទាល់ខ្លួន**។ សូមមើលតារាងទាំងបីខាងក្រោម។

**ក. កម្រិតភាគរយ — ដូចគ្នាគ្រប់កម្រិត ✓**

`floor(ពិន្ទុពេញ × 0.9 / 0.8 / 0.7 / 0.6 / 0.5)` ។ ផ្ទៀងផ្ទាត់ជាមួយឯកសារយោង៖

| ពិន្ទុពេញ | A | B | C | D | E |
|---|---|---|---|---|---|
| ១២៥ | ១១២ | ១០០ | ៨៧ | ៧៥ | ៦២ |
| ៧៥ | ៦៧ | ៦០ | ៥២ | ៤៥ | ៣៧ |
| ៥០ | ៤៥ | ៤០ | ៣៥ | ៣០ | ២៥ |
| ១០ (បឋម) | ៩ | ៨ | ៧ | ៦ | ៥ |

ត្រូវគ្នា ១០០%។ **ការបង្គត់គឺ `floor`** (112.5→112, 87.5→87, 62.5→62, 67.5→67, 52.5→52, 37.5→37) — កុំប្រើ `round`។

**ខ. ស្លាកនិទ្ទេស — ខុសគ្នា ដូច្នេះ `bands` នៅតែជា per-scheme**

| និទ្ទេស | បឋមសិក្សា (`scheme.ts` បច្ចុប្បន្ន) | អនុ/វិទ្យាល័យ |
|---|---|---|
| A | ល្អណាស់ | **ល្អប្រសើរ** |
| B | ល្អ | **ល្អណាស់** |
| C | ល្អបង្គួរ | **ល្អ** |
| D | — | **ល្អបង្គួរ** |
| E | — | **មធ្យម** |
| F | — | **ខ្សោយ** |

ស្លាក**រំកិលមួយកម្រិត**។ `gradeFor()` ត្រូវ normalise ទៅភាគរយ ហើយ **យក `bands` ពី scheme របស់កម្រិត** មិនមែនពី table ចែករួម។

**គ. បាក់ឌុប — តារាងផ្សេង មិនមែន ៩០/៨០/៧០**

| និទ្ទេស | ភាគរយផ្ទៃក្នុង | បាក់ឌុប (លើ ៥០០) |
|---|---|---|
| A | ៩០–១០០% | ៤២៧–៥០០ |
| B | ៨០–៨៩% | ៣៨០–៤២៦ |
| C | ៧០–៧៩% | ៣៣២–៣៧៩ |
| D | ៦០–៦៩% | ២៨៥–៣៣១ |
| E | ៥០–៥៩% | ២៣៧–២៨៤ |
| F | < ៥០% | ០–២៣៦ |

កម្រិតបាក់ឌុប = `floor(475 × k)` តែជួរខាងលើបញ្ចប់ត្រឹម ៥០០។ **កុំគណនាវា — encode វាជាតារាងផ្ទាល់** ព្រោះមូលដ្ឋាន (៤៧៥) និងអតិបរមា (៥០០) មិនដូចគ្នា ហើយប្រភពទាំងពីររបស់អ្នកយល់ស្របលើលេខទាំងនេះ។ ដូច្នេះ scheme ត្រូវការ band set ពីរ៖ ផ្ទៃក្នុង និងបាក់ឌុប។

### 3.3b ឯកសារយោងចាស់ (មិនប្រើទៀត)

ចំណុចនេះសំខាន់៖ ជញ្ជាំង A–F បច្ចុប្បន្នរបស់អ្នក (A≥9, B≥8, C≥7, D≥6, E≥5 លើ /១០) គឺ **៩០ / ៨០ / ៧០ / ៦០ / ៥០ ភាគរយ**។ មធ្យមភាគ /៥០ របស់អនុវិទ្យាល័យប្រើភាគរយ**ដូចគ្នាបេះបិទ** — A≥45, B≥40, C≥35, D≥30, E≥25។

ដូច្នេះកុំបង្កើត band table ទី២។ ប្តូរ `gradeFor()` ឲ្យ normalize ទៅភាគរយមុន match:

```ts
const pct = (average / config.maxScore) * 100
// match band តាម band.min / config.maxScore * 100
```

សម្រាប់បឋមសិក្សា (`maxScore: 10`) នេះ**មិនប្តូរឥរិយាបថទាល់តែសោះ** — 9/10 = 90% ដដែល។ ជា refactor សុទ្ធ។

### 3.4 អ្វីដែលត្រូវលុបចោល

`custom_subjects` (migration 00012) ត្រូវជំនួសដោយ row `scope='class', source='custom'`។ ធ្វើជា migration បម្លែងម្តង រួច `useCustomSubjects` ក្លាយជា wrapper លើ resolver ថ្មី។ កុំទុកឲ្យប្រព័ន្ធពីរដំណើរការជាមួយគ្នា — នោះជាកំហុសដដែលដែល scoping legacy/v2 កំពុងជួប។

## ៤. Resolver

ដាក់នៅ `lib/scores/template.ts` — pure, គ្មាន server-only import (ដូច `lib/rbac/permissions.ts`)។

```ts
export interface EffectiveSubject {
  subjectKey: string
  labelKm: string
  groupLabel: string | null
  maxScore: number          // មេគុណមកពី coefficientOf(maxScore, scheme)
  columns: ScoreColumn[]
  valueKind: 'numeric' | 'text'
  sortOrder: number
  origin: 'system' | 'school' | 'class'   // សម្រាប់បង្ហាញ badge "កែដោយគ្រូ"
}

export function resolveTemplate(
  rows: ScoreTemplateSubjectRow[],
  scoreType: ScoreType,
): EffectiveSubject[]
```

ក្បួន merge: group តាម `subject_key` → យក row ស្រទាប់ទាបបំផុត → បើ `hidden` ចោល → filter តាម `score_types` → តម្រៀបតាម `sort_order`។

រួច:
- **client** — hook `useScoreTemplate()` ស្របតាម `useCustomSubjects` ដែលមានស្រាប់
- **server** — `resolveServerTemplate()` ក្នុង `lib/utils/serverScope.ts` ព្រោះវាត្រូវការ `resolveServerScope()` រួចហើយ

### វិទ្យាល័យ ថ្នាក់ទី១០–១១ — មិនទាន់មាន

ថ្នាក់ទី១០ ដូចអនុវិទ្យាល័យ (`hs_*` prefix)។ **ទី១១–១២ បំបែកជាក្រុម** — នេះជាវិមាត្រថ្មីមួយទៀត៖

```
ថ្នាក់ទី១២ ─┬─ ក្រុមវិទ្យាសាស្ត្រ        → មុខវិជ្ជាជម្រើស: រូប/គីមី/ជីវ/ផែនដី
            └─ ក្រុមវិទ្យាសាស្ត្រសង្គម   → មុខវិជ្ជាជម្រើស: ប្រវត្តិ/ភូមិ/សីលធម៌/សេដ្ឋកិច្ច
```

មុខវិជ្ជាចាំបាច់ (គ្រប់ក្រុម): ភាសាខ្មែរ, គណិតវិទ្យា, ភាសាបរទេស, ប្រវត្តិវិទ្យា (ឬតាមរចនាសម្ព័ន្ធ BacII បច្ចុប្បន្ន)។

**ផលប៉ះពាល់លើ data model**: ក្រុមមិនមែនជាកម្រិត និងមិនមែនជាថ្នាក់ — គឺជា **variant នៃ template**។ ដំណោះស្រាយសាមញ្ញបំផុត៖ បន្ថែម `track TEXT` លើ `score_template_subjects` និង `classes.track`។ Resolver filter បន្ថែមមួយជួរ។ **កុំបង្កើតតារាងថ្មី** សម្រាប់រឿងនេះ។

- ព្រមានពេលពិន្ទុពេញមិនមែនពហុគុណនៃ ២៥ ("មេគុណ 1.36 — ប្រាកដទេ?") តែកុំរារាំង។
- មុខវិជ្ជាប្រព័ន្ធ **លាក់បាន តែលុបមិនបាន** — ព្រោះពិន្ទុចាស់នៅភ្ជាប់នឹង key នោះ។ លុបនឹងធ្វើឲ្យទិន្នន័យក្លាយជា orphan។
- ការប្តូរពិន្ទុពេញ **ប្តូរមេគុណ ដូច្នេះប្តូរមធ្យមភាគចាស់ទាំងអស់** — ត្រូវព្រមានច្បាស់ ឬចាក់សោបន្ទាប់ពីមានពិន្ទុរួច
- ប្រើ `SearchableSelect` និង `Dialog` ដែលមានស្រាប់ — កុំធ្វើ dropdown ថ្មី (តាម `CLAUDE.md`)
- រាល់ការកែ → `auditLog()` ព្រោះវាប៉ះលទ្ធផលសិស្ស

### កំណត់សម្គាល់អនុវត្ត (ខុសពី spec ត្រង់ណា ហេតុអ្វី)

* **§3.1 `education_level_id`/`grade_id` មិនអាចប្រើសម្រាប់ system rows** — តារាងទាំងពីរជា per-school (`school_id NOT NULL`), ដូច្នេះ national default គ្មាន UUID សកលឲ្យចង្អុល។ ការអនុវត្តបន្ថែម `level_key TEXT` (`primary|lower_secondary|upper_secondary`), `grade_number INT`, `track TEXT` ជា column សកលលើ `score_template_subjects` (00021)។ FK per-school នៅរក្សាសម្រាប់ school-scope narrowing។
* **§3.1 `source` column មិនបានបង្កើត** — ប្រភពស្មានចេញពីទិន្នន័យ: row ថ្នាក់ដែលគ្មាន system/school row ក្រោមខ្លួន = គ្រូបន្ថែម។ Flag ផ្ទុកអាចនឹង drift; ការស្មានមិនអាច។
* **`sem_behavior_all` ជា `value_kind='text'`** ព្រោះគ្រប់ column របស់វាជា dropdown ពាក្យខ្មែរ (មិនមែនតែ `sem_eval_*` បួនដាច់ដោយឡែក)។
* **Override ដែលលែងខុសពី parent ត្រូវលុប មិនមែនរក្សា** — row ស្តួចនៅតែ pin និយមន័យ ធ្វើឲ្យ default ថ្មីមិនដល់ថ្នាក់នោះ។

ដំណាក់ 0–1 គឺជា **refactor គ្មានការប្តូរឥរិយាបថ**។ សូមកុំ ship ដំណាក់ 2 មុនពេល 1 ដំណើរការស្ថិតស្ថេរ។

## ៩. អ្វីដែលខ្ញុំត្រូវការពីអ្នក

ខ្ញុំមិនចង់ដាក់លេខមិនច្បាស់ចូលក្នុងផលិតផលដែលគ្រូខ្មែរប្រើពិត។ សូមផ្តល់៖

✅ **ដោះស្រាយរួច**: ក្បួនមេគុណ = ពិន្ទុពេញ ÷ ៥០

នៅសល់៖

1. **ពិន្ទុពេញតាមមុខវិជ្ជា** សម្រាប់ ទី៧, ទី៨, ទី៩, ទី១០, ទី១១, ទី១២ — នេះជាអ្វីដែលនៅខ្វះតែមួយគត់ដើម្បីសរសេរ seed។ រូបថតតារាងស្រង់ពិន្ទុ MoEYS ក៏បានដែរ។
   *ខុសគ្នាតាមថ្នាក់ដែរឬទេ? ឧ. គណិតវិទ្យា ទី៧ និង ទី១២ ពិន្ទុពេញដូចគ្នាទេ?*

2. **ច្បាប់ "ដក ២៥"** ភាសាបរទេស ទី៩/ទី១២ — មួយណាត្រូវ?
   - (ក) ពិន្ទុសិស្សដក ២៥ ត្រង់ៗ (៦០ → ៣៥)
   - (ខ) ពិន្ទុពេញបន្ថយពី ៧៥ → ៥០ (មេគុណ 1.5 → 1) ហើយពិន្ទុ scale តាម
   - (គ) អ្វីផ្សេង
   លទ្ធផលខុសគ្នាឆ្ងាយណាស់ ដូច្នេះខ្ញុំមិនទាយទេ។

3. **មធ្យមភាគបង្ហាញជា /៥០ មែនទេ?** ឬសាលាបំលែងទៅ /១០ លើប្រតិវេទន៍? (ប៉ះពាល់ certificate និង parent-report)

4. **រចនាសម្ព័ន្ធក្រុម ទី១១–១២** — មុខវិជ្ជាចាំបាច់ vs ជម្រើស តាមក្រុមវិទ្យាសាស្ត្រ/សង្គម

5. **គំរូអាជីវកម្ម**: កម្រិតអនុ/វិទ្យាល័យជា feature Premium ឬអត់? (មាន `premium_requests` ស្រាប់)

ចំណុច ១ តែមួយគត់ដែលរារាំងដំណាក់កាល ២។ ដំណាក់កាល ០–១ ចាប់ផ្តើមបានឥឡូវនេះ។

### បញ្ហា ៣ ដែលត្រូវសម្រេចមុនសរសេរ code

**១. មុន login គ្មាន `auth.uid()` ដូច្នេះមិនអាចសរសេរចូល DB បាន**

Gmail OAuth នាំចេញពី app រួចត្រឡប់មកវិញ។ ការជ្រើសកម្រិតត្រូវរស់រានឆ្លងកាត់ការ redirect នោះ។ ដំណោះស្រាយ៖ ទុកជា **ជម្រើសបណ្តោះអាសន្ន** ក្នុង `sessionStorage` ឬ OAuth `state` រួច **សរសេរជាផ្លូវការក្រោយ login**។ អេក្រង់មុន login ជាទីផ្សារ មិនមែនជាប្រភពនៃការពិត។

**២. "ប្តូរកម្រិតមិនបានទេ" ជាការសន្យាដែលពិបាករក្សា**

ការហាមមានហេតុផលបច្ចេកទេសពិត៖ ប្តូរកម្រិត = ប្តូរ `subject_key` ទាំងស្រុង ហើយពិន្ទុចាស់ក្លាយជា orphan។

តែវាជាការសម្រេចចិត្តដែល**មិនអាចត្រឡប់វិញ ធ្វើនៅពេលអ្នកប្រើដឹងតិចបំផុត** — មុនសូម្បីតែឃើញ dashboard។ គ្រូនឹងចុចខុស។ អនុសាសន៍៖ ហាមមិនឲ្យ**គ្រូខ្លួនឯង**ប្តូរ តែត្រូវមាន**ផ្លូវសង្គ្រោះសម្រាប់ support** (គណនីថ្មីគ្មានពិន្ទុ = ប្តូរបានដោយសេរី; មានពិន្ទុរួច = ត្រូវការការអន្តរាគមន៍)។ សរសេរវាជា "ប្តូរដោយខ្លួនឯងមិនបានទេ" មិនមែន "ប្តូរមិនបានទេ"។

**៣. ស្វែងរកស្ថាប័នដើម្បីចូលរួម — RLS បច្ចុប្បន្នរារាំង**

`schools_select_member` អនុញ្ញាតឲ្យអានតែសាលាដែលអ្នកជាសមាជិកប៉ុណ្ណោះ។ ដូច្នេះ **គ្រូមិនអាចស្វែងរកសាលាដែលខ្លួនមិនទាន់ចូលរួមបានទេ** — feature នេះមិនអាចធ្វើបានក្រោម policy បច្ចុប្បន្ន។

ត្រូវការ ២ យ៉ាងថ្មី៖
- **ផ្លូវអានសាធារណៈ** — RPC ឬ view ដែលបង្ហាញតែ `id`, `name`, ខេត្ត/ស្រុក។ **កុំបង្ហាញចំនួនសិស្ស ឬឈ្មោះគ្រូ** — នោះជាការលេចធ្លាយទិន្នន័យ។
- **ការស្នើសុំ និងការអនុម័ត** — ការចូលរួមដោយស្វ័យប្រវត្តិមានន័យថាអ្នកណាក៏អាចចូលសាលាណាក៏បាន ហើយឃើញទិន្នន័យសិស្ស។ ត្រូវតែ `join_requests` → នាយកអនុម័ត។

នេះជា feature ដាច់ដោយឡែកមួយ ធំជាងការប្តូរលំដាប់អេក្រង់។

### ផលប៉ះពាល់លើ template

កម្រិតកំណត់ `education_level_id` ដែល resolver ប្រើរួច (§២)។ ដូច្នេះ flow នេះមិនប្តូរ data model ទេ — វាគ្រាន់តែ**កំណត់តម្លៃមួយឲ្យឆាប់ជាងមុន**។ ការងារពិតគឺ seed template តាមកម្រិត មិនមែន resolver។
